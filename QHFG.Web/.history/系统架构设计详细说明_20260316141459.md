### 1. **请求封装层（axios.js）**

```javascript
// src/request/axios.js
import axios from "axios";

const baseURL = "http://129.211.174.149:18080";
const request = axios.create({
  baseURL: baseURL,
  timeout: 100000,
});

// 请求拦截器：统一处理请求
request.interceptors.request.use(
  (config) => {
    // 自动注入Token
    const userInfo = localStorage.getItem("leiyangUser");
    let token = "";
    if (userInfo) {
      try {
        const user = JSON.parse(userInfo);
        token = user.token;
      } catch (e) {
        console.error("解析用户信息失败:", e);
      }
    }
    
    config.headers = {
      ...config.headers,
      "Content-Type": "application/json;charset=UTF-8",
      token: token, // 自动添加token
    };
    
    return config;
  },
  (error) => {
    return Promise.reject(error);
  }
);

// 响应拦截器：统一处理响应
request.interceptors.response.use(
  (res) => {
    // Token自动更新
    if (res && res.data) {
      if (res.data.token) {
        const userInfo = localStorage.getItem("leiyangUser");
        if (userInfo) {
          try {
            const user = JSON.parse(userInfo);
            user.token = res.data.token;
            localStorage.setItem("leiyangUser", JSON.stringify(user));
          } catch (e) {
            console.error("更新 token 失败:", e);
          }
        }
      }
      return res.data;
    }
    throw Error("请求数据失败");
  },
  (err) => {
    // 统一错误处理
    if (err.response && err.response.data) {
      return Promise.resolve(err.response.data);
    }
    return Promise.reject(err);
  }
);

export default request;
```

**数据层职责**：
- ✅ **统一封装**：封装axios，统一配置
- ✅ **自动处理**：Token自动注入和更新
- ✅ **错误处理**：统一错误处理逻辑
- ✅ **响应格式化**：统一响应数据格式

### 3. **文件下载工具（downloadFile.js）**

```javascript
// src/utils/downloadFile.js
// 获取blob
export function handleGetBlob(url) {
  return new Promise(resolve => {
    const xhr = new XMLHttpRequest();
    xhr.open("GET", url, true);
    xhr.responseType = "blob";
    xhr.onload = () => {
      if (xhr.status === 200) {
        resolve(xhr.response);
      }
    };
    xhr.setRequestHeader("Cache-Control", "no-cache");
    xhr.send();
  });
}

// 下载文件并重新命名
export function downloadFile(url, name) {
  return new Promise((resolve, reject) => {
    handleGetBlob(url)
      .then(blob => {
        let fileName = url.split("/");
        fileName = fileName[fileName.length - 1];
        
        let fileType = fileName.split(".");
        fileType = fileType[fileType.length - 1];
        
        let saveName = `${name}.${fileType}`;
        handleSaveAs(blob, saveName);
        resolve();
      })
      .catch(err => {
        reject(err);
      });
  });
}

// 保存文件
export function handleSaveAs(blob, filename) {
  if (window.navigator.msSaveOrOpenBlob) {
    navigator.msSaveBlob(blob, filename);
  } else {
    const link = document.createElement("a");
    const body = document.querySelector("body");
    link.href = window.URL.createObjectURL(blob);
    link.download = filename;
    link.style.display = "none";
    body.appendChild(link);
    link.click();
    body.removeChild(link);
    window.URL.revokeObjectURL(link.href);
  }
}
```

- 支持跨模块复用

### ✅ 可维护
- 代码结构清晰，易于定位问题
- 修改影响范围小，降低维护成本
